with prod_code_data as (
--《V3 SPC监控for北极星-V2》王书
select distinct B.product_code 
,A.productspecname 
from EDA.IMP_SPC_TZBJX A 
left join DWR_MES_PRODUCTSPEC_V B on A.productspecname =B.prod_id 
where 1=1
${IF(LEN(cmcbProdid)=0,"","AND B.product_code IN ('"+cmcbProdid+"')")}  
)
,spc_raw_data as (
	select 'ARRAY'::varchar(100) as factory
	,a.step_id 
	,(a.step_id ||'-'|| B.description)::varchar(200) as OPER_NAME
	,a.sheet_id as mat_id
	,a.sheet_start_time as start_time
	,to_char(a.sheet_start_time,'YYYYMMDD')::varchar(100) as date_timekey
	,a.product_spec 
	,a.equip_id 
	,a.pre_sub_unit_id
	,a.param_name 
	,c.product_code::varchar(100) as prod_code
	,a.site_name 
	,a.param_value 
	,(dense_rank ()OVER(partition by A.sheet_id,A.step_id,A.param_name order by a.sheet_start_time desc))::numeric as rn
	from eda.spc_tzbjx_array a
	left join dwr_mes_processoperationspec_v B on A.step_id =B.oper_code
	left join dwr_mes_productspec_v c on a.product_spec =c.prod_id 
	inner join prod_code_data s on a.product_spec =s.productspecname
	inner join EDA.IMP_SPC_TZBJX x on a.step_id =x.step_id and a.param_name =x.parmtername and x.data_type ='SPC'
	where 1=1
	and A.sheet_start_time >='${dtStartDate}'::TIMESTAMP
	and A.sheet_start_time <='${dtEndDate}'::TIMESTAMP + interval '1day'
	and B.detail_oper_code_type not in ('VIEW','AOI')
	and abs(a.param_value) !=299.99
	${IF(LEN(cmcbStep)=0,"","AND A.step_id IN ('"+cmcbStep+"')")}
	${IF(LEN(cmcbParamname)=0,"","AND a.param_name IN ('"+cmcbParamname+"')")}
	${IF(LEN(cmcbFactory)=0,"","AND B.factory IN ('"+cmcbFactory+"')")}
	union all
	select 'OLED'::varchar(100) as factory
	,a.step_id 
	,(a.step_id ||'-'|| B.description)::varchar(200) as OPER_NAME
	,a.glass_id  as mat_id
	,a.glass_start_time  as start_time
	,to_char(a.glass_start_time,'YYYYMMDD')::varchar(100) as date_timekey
	,a.product_spec 
	,a.equip_id 
	,a.pre_sub_unit_id
	,a.param_name 
--	,c.product_code as prod_code
	,(case when c.product_code ='MQC' then c.product_code ||'-'|| coalesce (G.PRODCODE,'') else c.product_code end)::varchar(100) as prod_code
	,a.site_name 
	,a.param_value 
	,(dense_rank ()OVER(partition by A.glass_id,A.step_id,A.param_name order by a.glass_start_time desc))::numeric  as rn
	from eda.spc_tzbjx_oled a
	left join dwr_mes_processoperationspec_v B on A.step_id =B.oper_code 
	left join dwr_mes_productspec_v c on a.product_spec =c.prod_id 
	inner join prod_code_data s on a.product_spec =s.productspecname
	left join dwd_calendar H on to_char(a.glass_start_time,'YYYYMMDD') = H.time_key and H.factory ='ARRAY'
	left join (select distinct ppid ,prodcode from EDA.IMP_SPC_TZBJX_MQC) G on A.PPID =G.PPID and c.product_code ='MQC' 
	inner join EDA.IMP_SPC_TZBJX x on a.step_id =x.step_id and a.param_name =x.parmtername and x.data_type ='SPC'
	where 1=1
	and A.glass_start_time >='${dtStartDate}'::TIMESTAMP
	and A.glass_start_time <='${dtEndDate}'::TIMESTAMP + interval '1day'
	and B.detail_oper_code_type not in ('VIEW','AOI')
	and abs(a.param_value) !=299.99
	and case when '${cmcbProdid}' ='MQC' then (c.product_code ||'-'|| coalesce (G.PRODCODE,''))::varchar(100) = '${cmcbMqcProd}' else 1=1 end 
	${IF(LEN(cmcbStep)=0,"","AND A.step_id IN ('"+cmcbStep+"')")}
	${IF(LEN(cmcbParamname)=0,"","AND a.param_name IN ('"+cmcbParamname+"')")}
	${IF(LEN(cmcbFactory)=0,"","AND B.factory IN ('"+cmcbFactory+"')")}
	and case when (a.step_id ='21230') AND (a.param_name='B_0_CIE_Y') then (a.param_value>0.0415) and (a.param_value<0.0555) 
		when (a.step_id ='21230') AND (a.param_name='Bi_0') then (a.param_value>114) and (a.param_value<210) 
		when (a.step_id ='21230') AND (a.param_name='G_0_CIE_X') then (a.param_value>0.2161) and (a.param_value<0.2761) 
		when (a.step_id ='21230') AND (a.param_name='G_0_E') then (a.param_value>127) and (a.param_value<223) 
		when (a.step_id ='21230') AND (a.param_name='JNCD_30') then (a.param_value>0) and (a.param_value<100) 
		when (a.step_id ='21230') AND (a.param_name='JNCD_45') then (a.param_value>0) and (a.param_value<100) 
		when (a.step_id ='21230') AND (a.param_name='JNCD_60') then (a.param_value>0) and (a.param_value<100) 
		when (a.step_id ='21230') AND (a.param_name='R_0_CIE_X') then (a.param_value>0.6721) and (a.param_value<0.6861) 
		when (a.step_id ='21230') AND (a.param_name='R_0_E') then (a.param_value>40) and (a.param_value<100)
		when (a.step_id ='21200') AND (a.param_name in ('PPA_B_X'
			,'PPA_B_Y'
			,'PPA_G1_X'
			,'PPA_G1_Y'
			,'PPA_G_X'
			,'PPA_G_Y'
			,'PPA_R1_X'
			,'PPA_R1_Y'
			,'PPA_R_X'
			,'PPA_R_Y')) then (a.param_value>=-6) and (a.param_value<=6)
		ELSE 1=1
		end
	union all
	select 'TP'::varchar(100) as factory
	,a.step_id 
	,(a.step_id ||'-'|| B.description)::varchar(200) as OPER_NAME
	,a.glass_id  as mat_id
	,a.glass_start_time  as start_time
	,to_char(a.glass_start_time,'YYYYMMDD')::varchar(100) as date_timekey
	,a.product_spec 
	,a.equip_id 
	,a.pre_sub_unit_id
	,a.param_name 
	,c.product_code::varchar(100) as prod_code
	,a.site_name 
	,a.param_value
	,(dense_rank ()OVER(partition by A.glass_id,A.step_id,A.param_name order by a.glass_start_time desc))::numeric as rn
	from eda.spc_tzbjx_tsp a
	left join dwr_mes_processoperationspec_v B on A.step_id =B.oper_code 
	left join dwr_mes_productspec_v c on a.product_spec =c.prod_id 
	inner join prod_code_data s on a.product_spec =s.productspecname
	inner join EDA.IMP_SPC_TZBJX x on a.step_id =x.step_id and a.param_name =x.parmtername and x.data_type ='SPC'
	where 1=1
	and A.glass_start_time >='${dtStartDate}'::TIMESTAMP
	and A.glass_start_time <='${dtEndDate}'::TIMESTAMP + interval '1day'
	and B.detail_oper_code_type not in ('VIEW','AOI')
	and abs(a.param_value) !=299.99
	${IF(LEN(cmcbStep)=0,"","AND A.step_id IN ('"+cmcbStep+"')")}
	${IF(LEN(cmcbParamname)=0,"","AND a.param_name IN ('"+cmcbParamname+"')")}
	${IF(LEN(cmcbFactory)=0,"","AND B.factory IN ('"+cmcbFactory+"')")}
) 
,FANEL_DATA as (
select a.mat_id
,a.factory
,a.OPER_NAME 
,a.step_id 
,a.prod_code
,a.param_name
,A.PARAM_VALUE
,coalesce (c.main_step_id,a.step_id ) as main_step_id
,coalesce (c.main_eqp_type,'EQP') as main_eqp_type
,case when coalesce (c.main_eqp_type,'EQP') ='EQP' and a.factory ='ARRAY' then coalesce(D.EQP_ID ,A.equip_id )::varchar(100)
	when coalesce (c.main_eqp_type,'EQP') ='CHAMBER' and a.factory ='ARRAY' then coalesce(E.SUB_UNIT_ID,F.UNIT_ID)::varchar(100)
	when coalesce (c.main_eqp_type,'EQP') ='EQP' and a.factory in ('OLED','TP') then coalesce(G.EQP_ID ,A.equip_id )::varchar(100)
	when coalesce (c.main_eqp_type,'EQP') ='CHAMBER' and a.factory in ('TP') then H.SUB_UNIT_ID::varchar(100)
	when coalesce (c.main_eqp_type,'EQP') ='CHAMBER' and a.factory in ('OLED') then J.SUB_UNIT_ID::varchar(100) end as EQ_ID
--,case when coalesce (c.main_eqp_type,'EQP') ='EQP' and a.factory ='ARRAY' then coalesce(D.event_timekey ,to_char(a.start_time,'YYYYMMDDHH24miss')) 
--	when coalesce (c.main_eqp_type,'EQP') ='CHAMBER' and a.factory ='ARRAY' then coalesce(E.event_timekey,F.event_timekey)
--	when coalesce (c.main_eqp_type,'EQP') ='EQP' and a.factory in ('OLED','TP') then coalesce(G.event_timekey ,to_char(a.start_time,'YYYYMMDDHH24miss') ) 
--	when coalesce (c.main_eqp_type,'EQP') ='CHAMBER' and a.factory in ('OLED','TP') then H.event_timekey end as EQ_TIMEKEY
,to_char(a.start_time,'YYYYMMDDHH24miss') as EQ_TIMEKEY
from spc_raw_data a
left join dwd_imp_dv_param_spec c on a.prod_code =c.prod_code and a.step_id = c.step_id and a.param_name =c.param_name
left join DWT_INOUT_SHT D on A.mat_id = D.sheet_id and coalesce (c.main_step_id,a.step_id )::varchar(200) =D.OPER_CODE
							AND D.INOUT_TYPE = 'OUT'
							and D.DATE_TIMEKEY >= TO_CHAR('${dtStartDate}'::TIMESTAMP - INTERVAL '1 MONTH','YYYYMMDD')
		   					AND D.DATE_TIMEKEY <= REPLACE('${dtEndDate}','-','')
		   					and case when '${cmcbFactory}'='ARRAY'::text  then 1=1 else 1=2 end 
left join DWT_INOUT_SUB_UNIT_SHT E on A.mat_id = E.sheet_id and coalesce (c.main_step_id,a.step_id ) = E.oper_code  
							AND SPLIT_PART(E.SUB_UNIT_ID,'-',2)::TEXT IN ('CVD'::text,'SPU'::text,'DRE'::text,'OVE'::text)
           					AND SUBSTR(E.SUB_UNIT_ID,8,6)::TEXT IN ('DRE-PC'::text,'CVD-CH'::text,'OVE-CH'::text,'SPU-PM'::text)
           					AND E.INOUT_TYPE = 'OUT'
           					and e.DATE_TIMEKEY >= TO_CHAR('${dtStartDate}'::TIMESTAMP - INTERVAL '1 MONTH','YYYYMMDD')
		   					AND e.DATE_TIMEKEY <= REPLACE('${dtEndDate}','-','')
		   					and case when '${cmcbFactory}'='ARRAY'::text then 1=1 else 1=2 end 
left join DWT_INOUT_UNIT_SHT F on  A.mat_id = F.sheet_id and coalesce (c.main_step_id,a.step_id )::varchar(200) = F.oper_code 
							AND SUBSTR(F.UNIT_ID,8,2) IN ('CH'::text)  
							AND F.INOUT_TYPE = 'OUT'
							and f.DATE_TIMEKEY >= TO_CHAR('${dtStartDate}'::TIMESTAMP - INTERVAL '1 MONTH','YYYYMMDD')
		   					AND f.DATE_TIMEKEY <= REPLACE('${dtEndDate}','-','')
		   					and case when '${cmcbFactory}'='ARRAY'::text then 1=1 else 1=2 end 
left join DWT_INOUT_GLS G on A.mat_id = G.glass_id and coalesce (c.main_step_id,a.step_id )::varchar(200) = G.oper_code
							AND G.INOUT_TYPE = 'OUT'
							and g.DATE_TIMEKEY >= TO_CHAR('${dtStartDate}'::TIMESTAMP - INTERVAL '1 MONTH','YYYYMMDD')
		   					AND g.DATE_TIMEKEY <= REPLACE('${dtEndDate}','-','')
		   					and case when '${cmcbFactory}'='OLED'::text or '${cmcbFactory}'='TP'::text then 1=1 else 1=2 end 
left join DWT_INOUT_SUB_UNIT_GLS H on A.mat_id = H.glass_id and coalesce (c.main_step_id,a.step_id )::varchar(200) = H.oper_code
							AND H.INOUT_TYPE = 'OUT'
						    AND H.FACTORY = 'TP'
				            AND SPLIT_PART(H.SUB_UNIT_ID,'-',2)::TEXT IN ('CVD'::text,'SPU'::text,'DRE'::text,'OVE'::text)
				            AND SUBSTR(H.SUB_UNIT_ID,8,6)::TEXT IN ('DRE-PC'::text,'CVD-CH'::text,'OVE-CH'::text,'SPU-PM'::text)
				            and h.DATE_TIMEKEY >= TO_CHAR('${dtStartDate}'::TIMESTAMP - INTERVAL '1 MONTH','YYYYMMDD')
		   					AND h.DATE_TIMEKEY <= REPLACE('${dtEndDate}','-','')
		   					and case when '${cmcbFactory}'='TP'::text then 1=1 else 1=2 end 
left join (select glass_id 
				,O.new_oper as oper_code 
				,case when O.new_oper::TEXT in ('21200-CVD1'::text,'21200-CVD2'::text,'21200-CVD3'::text,'21200-CVD4'::text) AND substring(o.sub_unit_id,12,2)::text ='CH'::TEXT then '21200-CVD'::varchar(100) else O.new_oper  end as OPER_CODE_NEW
				,J.eqp_id 
				,J.sub_unit_id 
				from DWT_INOUT_SUB_UNIT_GLS J
				INNER JOIN DWD_MES_OLED_OPER_LAYER_V O
				         ON J.OPER_CODE = O.OPER_CODE 
				         and case when '${cmcbFactory}'::text='OLED'::text then 1=1 else 1=2 end
				         AND J.SUB_UNIT_ID = O.SUB_UNIT_ID 
				         AND O.NEW_OPER IS NOT null
						 AND j.INOUT_TYPE = 'OUT'
						 AND j.FACTORY LIKE 'OLED%'
				where 1=1
				and J.DATE_TIMEKEY >= TO_CHAR('${dtStartDate}'::TIMESTAMP - INTERVAL '1 MONTH','YYYYMMDD')
				AND J.DATE_TIMEKEY <= REPLACE('${dtEndDate}','-','')
) J on A.mat_id = J.glass_id and (coalesce (c.main_step_id,a.step_id )::varchar(200) = J.oper_code or coalesce (c.main_step_id,a.step_id )::varchar(200) = J.OPER_CODE_NEW)
where 1=1
and rn =1
)
select B.prod_code
	,B.step_id
	,main_step_id
	,EQ_ID
	,(EQ_ID ||' '|| EQ_TIMEKEY||' '||mat_id)::varchar(100) as x_type
	,B.PARAM_NAME
	,case when max_value > q3_value + 1.5 * (q3_value -q1_value) then q3_value + 1.5 * (q3_value -q1_value) else max_value end as max_value
	,q3_value
	,median_value
	,q1_value
	,case when min_value < q1_value - 1.5 * (q3_value -q1_value) then q1_value - 1.5 * (q3_value -q1_value) else min_value end as min_value
from (
	select a.prod_code
	,a.step_id
	,A.main_step_id
	,coalesce (EQ_ID,'-')::varchar(100) as EQ_ID
	,EQ_TIMEKEY
	,mat_id
	--,EQ_ID ||' '|| EQ_TIMEKEY as x_type
	,A.PARAM_NAME
	,percentile_cont(0.25) WITHIN GROUP (ORDER BY PARAM_VALUE) AS q1_value
	,percentile_cont(0.5) WITHIN GROUP (ORDER BY PARAM_VALUE) AS median_value
	,percentile_cont(0.75) WITHIN GROUP (ORDER BY PARAM_VALUE) AS q3_value
	,max(PARAM_VALUE) as max_value
	,min(PARAM_VALUE) as min_value
	from FANEL_DATA A
	group by a.prod_code
	,a.step_id
	,A.main_step_id
	,coalesce (EQ_ID,'-')::varchar(100)
	,EQ_TIMEKEY
	,A.PARAM_NAME
	,mat_id
) B
order by EQ_ID
,EQ_TIMEKEY
